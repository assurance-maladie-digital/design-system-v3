import { Meta } from '@storybook/addon-docs';
import qualityData from './code-quality-status.json';
import '@/stories/styles/shared.css';
import { useState } from 'react';

<Meta title="Composants/Statut des composants" />

<div className="header">
  <h1>Qualité code des composants Studio Design</h1>
</div>

<h2>Critères de qualité des composants</h2>

<table className="quality-criteria-table">
  <thead>
    <tr>
      <th>Critère</th>
      <th>Description</th>
      <th>Valeurs possibles</th>
    </tr>
  </thead>

  <tbody>
    <tr>
  <td>Props / Slots documentés</td>
  <td>Vérifie que les props publiques, slots et événements sont décrits dans la documentation Storybook.</td>
  <td>Documenté / Manquant</td>
</tr>

<tr>
  <td>Onglet source code</td>
  <td>Vérifie que le composant expose son code source Template + Script ou un exemple clair d’implémentation.</td>
  <td>Présent / Absent</td>
</tr>

<tr>
  <td>Stories requises</td>
  <td>Vérifie la présence des stories importantes : Default, Disabled, Validation.</td>
  <td>Complètes / Partiel / Non concerné </td>
</tr>

<tr>
  <td>Page usages UX</td>
  <td>Vérifie la présence d’une page Usages rédigés par les UX designers dédiée au composant.</td>
  <td>Présente / Absente</td>
</tr>

<tr>
  <td>Thème visuel</td>
  <td>Vérifie les modes visuels Dark et Light.</td>
  <td>Light / Dark</td>
</tr>

<tr>
  <td>Thème mode</td>
  <td>Vérifie que le composant possède les déclinaisons graphiques nécessaires.</td>
  <td>CNAM / ameli pro</td>
</tr>

<tr>
  <td>Playground interactif</td>
  <td>Vérifie la présence d’un playground ou d’exemples interactifs permettant de manipuler les props.</td>
  <td>Présent / Absent</td>
</tr>

<tr>
  <td>Tests</td>
  <td>Tests disponibles pour le composant.</td>
  <td>UT / A11y / Cypress</td>
</tr>

<tr>
  <td>Criticité</td>
  <td>Identifie les éléments bloquants ou incomplets impactant la qualité globale du composant.</td>
  <td>RAS / Anomalies détectées</td>
</tr>
  </tbody>
</table>

{(() => {
  const [searchTerm, setSearchTerm] = useState('');
  const [selectedCategory, setSelectedCategory] = useState('Toutes');

const components = qualityData.results
  .filter(Boolean)
  .filter(item =>
    !item.componentName?.toLowerCase().startsWith('amelipro')
    && !item.storybookTitle?.toLowerCase().includes('amelipro')
    && item.category?.toLowerCase() !== 'amelipro'
  );

  const categories = [
  'Toutes',
  ...new Set(
    components
      .map(item => item.category)
      .filter(category =>
        category &&
        category.toLowerCase() !== 'amelipro'
      )
  ),
];

  const filteredComponents = components.filter((item) => {
    const matchesSearch =
      searchTerm === ''
      || item.componentName.toLowerCase().includes(searchTerm.toLowerCase());

    const matchesCategory =
      selectedCategory === 'Toutes'
      || item.category === selectedCategory;

    return matchesSearch && matchesCategory;
  });

  const getTagClass = (value) => {
    if (value === true || value === 'Complet' || value === 'Présent') return 'tag-green';
    if (value === false || value === 'Manquant' || value === 'Absent') return 'tag-red';
    if (value === 'Partiel' || value === 'À compléter') return 'tag-blue';
    return 'tag-gray';
  };

  const renderBooleanTag = (value, trueLabel = 'Présent', falseLabel = 'Absent') => (
    <span className={`carbon-tag ${getTagClass(value)}`}>
      {value ? trueLabel : falseLabel}
    </span>
  );

  return (
    <div className="carbon-a11y-container">
      <div className="quality-filters">
        <label className="filter-field">
          <span>Catégorie</span>
          <select
            className="filter-input"
            value={selectedCategory}
            onChange={(event) => setSelectedCategory(event.target.value)}
          >
            {categories.map(category => (
              <option key={category} value={category}>
                {category}
              </option>
            ))}
          </select>
        </label>

        <label className="filter-field filter-search">
          <span>Recherche</span>
          <input
            type="text"
            className="filter-input"
            placeholder="Rechercher un composant..."
            value={searchTerm}
            onChange={(event) => setSearchTerm(event.target.value)}
          />
        </label>

        <div className="filter-count">
          {filteredComponents.length} composant{filteredComponents.length > 1 ? 's' : ''}
        </div>
      </div>

      <div className="table-wrapper">
        <table className="carbon-table">
          <thead>
            <tr>
              <th>Composant</th>
              <th>Props / Slots documentés</th>
              <th>Onglet source code</th>
              <th>Stories requises</th>
              <th>Page usages UX</th>
              <th>Theme visuel</th>
              <th>Theme mode</th>
              <th>Playground interactif</th>
              <th>Tests</th>
              <th>Criticité</th>
            </tr>
          </thead>

          <tbody>
            {filteredComponents.map((item) => {
              const storyPath = item.storybookTitle
                ?.toLowerCase()
                .replaceAll('/', '-')
                .replaceAll(' ', '');

              const compteAmeliLink =
                `${window.location.origin}/?path=/docs/${storyPath}--docs`;

              const ameliProLink =
                `${window.location.origin}/?path=/docs/${storyPath}--docs&globals=theme:ap2026`;

              return (
                <tr key={item.componentName}>
                  <td>{item.componentName}</td>

                  <td>
	<span className={`carbon-tag ${getTagClass(item.hasPropsDocumentation)}`}>
		{item.propsDocumentationLabel || (item.hasPropsDocumentation ? 'Documenté' : 'Manquant')}
	</span>
</td>

                  <td>{renderBooleanTag(item.hasSourceTab, 'Présent', 'Absent')}</td>

                  <td>
                    <span className={`carbon-tag ${getTagClass(item.requiredStoriesStatus)}`}>
                      {item.requiredStoriesStatus}
                    </span>
                  </td>

                  <td>{renderBooleanTag(item.hasUxUsagePage, 'Présente', 'Absente')}</td>
  <td>
                    <span className={`carbon-tag ${getTagClass(item.themeStatus)}`}>
                      {item.themeStatus}
                    </span>
                  </td>
                  <td>
                    <div className="theme-links">
                      <a
                        href={compteAmeliLink}
                        target="_blank"
                        rel="noopener noreferrer"
                        className="theme-link"
                      >
                        CNAM
                      </a>

                      <a
                        href={ameliProLink}
                        target="_blank"
                        rel="noopener noreferrer"
                        className="theme-link"
                      >
                        ameli pro
                      </a>
                    </div>
                  </td>

                

                  <td>{renderBooleanTag(item.hasInteractivePlayground, 'Présent', 'Absent')}</td>

                  <td>
                    <div className="tests-column">
                      <span className={item.hasUnitTest ? 'test-ok' : 'test-ko'}>UT</span>
                      <span className={item.hasA11yTest ? 'test-ok' : 'test-ko'}>A11y</span>
                      <span className={item.hasCypressTest ? 'test-ok' : 'test-ko'}>Cypress</span>
                    </div>
                  </td>

                  <td>
                    {item.criticality ? (
                      <span className="carbon-tag tag-red">{item.criticality}</span>
                    ) : (
                      <span className="carbon-tag tag-green">Aucune</span>
                    )}
                  </td>
                </tr>
              );
            })}
          </tbody>
        </table>
      </div>
    </div>
  );
})()}
