import infoData from './component-info.json';
import '../styles/shared.css';
import { useState } from 'react';

<Meta title="Démarrer/Suivi des composants" />

<style>
{`

<div className="header">
  <h1>Suivi des composants</h1>
  <p>
    Pour chaque composant : sa page, son statut, ses dernières versions fonctionnelle et accessibilité,
    et ses derniers commits regroupés par type (fonctionnel / accessibilité).
  </p>
</div>

{(() => {
  const [search, setSearch] = useState('');
  const [versionFilter, setVersionFilter] = useState('');
  const [a11yVersionFilter, setA11yVersionFilter] = useState('');
  const [statusFilter, setStatusFilter] = useState('');
  const [activeTab, setActiveTab] = useState('functional');

  // Versions fonctionnelles distinctes, triées de la plus récente à la plus ancienne.
  const versions = [...new Set(infoData.results.map((r) => r.functionalVersion).filter(Boolean))].sort(
    (a, b) => b.localeCompare(a, undefined, { numeric: true }),
  );

  // Versions accessibilité distinctes, triées de la plus récente à la plus ancienne.
  const a11yVersions = [...new Set(infoData.results.map((r) => r.a11yVersion).filter(Boolean))].sort(
    (a, b) => b.localeCompare(a, undefined, { numeric: true }),
  );

  const slug = (title) =>
    title.toLowerCase().replace(/[\s/'’()&,.]+/g, '-').replace(/^-+|-+$/g, '');

  const getUrl = (componentName, storybookTitle) =>
    storybookTitle
      ? `/?path=/docs/${slug(storybookTitle)}--docs`
      : `/?path=/docs/composants-${componentName.toLowerCase().replace(/[^a-z0-9]/g, '')}--docs`;

  const REPO = 'https://github.com/assurance-maladie-digital/design-system-v3';

  // Formate une date en français : "JJ/MM/AAAA" ou ISO "YYYY-MM-DD ..." → "25 mars 2026".
  const formatDate = (dateStr) => {
    if (!dateStr) return null;
    let date;
    if (/^\d{2}\/\d{2}\/\d{4}$/.test(dateStr)) {
      const [day, month, year] = dateStr.split('/');
      date = new Date(`${year}-${month}-${day}T00:00:00`);
    } else {
      date = new Date(dateStr);
    }
    if (Number.isNaN(date.getTime())) return dateStr;
    return date.toLocaleDateString('fr-FR', { day: 'numeric', month: 'long', year: 'numeric' });
  };

  const renderMessage = (msg) => {
    const m = msg.match(/#(\d+)/);
    if (!m) return msg;
    return (
      <>
        {msg.slice(0, m.index)}
        <a href={`${REPO}/pull/${m[1]}`} target="_blank" rel="noopener noreferrer">
          #{m[1]}
        </a>
        {msg.slice(m.index + m[0].length)}
      </>
    );
  };

  const renderCommitList = (commits) => {
    if (!commits.length) return <p className="ci-empty">Aucun commit à afficher.</p>;
    return (
      <ul className="ci-commits">
        {commits.map((c, i) => (
          <li key={i}>
            <span className="c-date">{formatDate(c.date)}</span>
            <span className="c-msg">{renderMessage(c.message)}</span>
          </li>
        ))}
      </ul>
    );
  };

  const rows = infoData.results.filter(
    (item) =>
      (search === '' || item.componentName.toLowerCase().includes(search.toLowerCase())) &&
      (versionFilter === '' || item.functionalVersion === versionFilter) &&
      (a11yVersionFilter === '' || item.a11yVersion === a11yVersionFilter) &&
      (statusFilter === '' || item.status === statusFilter),
  );

  const tabClass = (tab) => `ci-tab ${activeTab === tab ? 'active' : ''} ${tab === 'a11y' ? 'a11y' : ''}`;

  return (
    <div className="ci-container">
      <div className="ci-meta">
        <strong>{infoData.totalCount}</strong> composants · Dernière génération&nbsp;:{' '}
        {new Date(infoData.date).toLocaleDateString('fr-FR', { year: 'numeric', month: 'long', day: 'numeric' })}
      </div>

      <div className="ci-filters">
        <input
          type="text"
          className="ci-search"
          placeholder="Rechercher un composant..."
          value={search}
          onChange={(e) => setSearch(e.target.value)}
        />
        <select
          className="ci-select"
          value={versionFilter}
          onChange={(e) => setVersionFilter(e.target.value)}
          aria-label="Filtrer par version fonctionnelle"
        >
          <option value="">Toutes les versions fonctionnelles</option>
          {versions.map((v) => (
            <option key={v} value={v}>
              v{v}
            </option>
          ))}
        </select>
        <select
          className="ci-select"
          value={a11yVersionFilter}
          onChange={(e) => setA11yVersionFilter(e.target.value)}
          aria-label="Filtrer par version accessibilité"
        >
          <option value="">Toutes les versions a11y</option>
          {a11yVersions.map((v) => (
            <option key={v} value={v}>
              v{v}
            </option>
          ))}
        </select>
        <select
          className="ci-select"
          value={statusFilter}
          onChange={(e) => setStatusFilter(e.target.value)}
          aria-label="Filtrer par statut"
        >
          <option value="">Tous les statuts</option>
          <option value="actif">Actif</option>
          <option value="déprécié">Déprécié</option>
        </select>
      </div>

      <div className="ci-grid">
        {rows.map((item) => {
          const isFunctional = activeTab === 'functional';
          const version = isFunctional ? item.functionalVersion : item.a11yVersion;
          const date = isFunctional ? item.functionalDate : item.a11yDate;
          const commits = isFunctional ? item.commits || [] : item.a11yCommits || [];

          return (
            <div className="ci-card" key={item.componentName}>
              <div className="ci-card-header">
                <span className="ci-card-name">
                  <a href={getUrl(item.componentName, item.storybookTitle)}>
                    {item.componentName.split('/').pop()}
                  </a>
                </span>
                <span className={`ci-card-status ${item.status === 'déprécié' ? 'deprecie' : 'actif'}`}>
                  <span className={`ci-card-dot ${item.status === 'déprécié' ? 'deprecie' : 'actif'}`}></span>
                  {item.status === 'déprécié' ? 'Déprécié' : 'Actif'}
                </span>
              </div>
              <div className="ci-tabs">
                <button
                  className={tabClass('functional')}
                  onClick={() => setActiveTab('functional')}
                  aria-pressed={isFunctional}
                >
                  Fonctionnel
                </button>
                <button
                  className={tabClass('a11y')}
                  onClick={() => setActiveTab('a11y')}
                  aria-pressed={!isFunctional}
                >
                  Accessibilité
                </button>
              </div>
              <div className="ci-card-body">
                <div className="ci-version-line">
                  {version ? (
                    <span className={`ci-tag ${isFunctional ? 'func' : 'a11y'}`}>
                      v{version}
                    </span>
                  ) : (
                    <span className="ci-tag empty">Version non renseignée</span>
                  )}
                  {version && <span className="ci-date">{formatDate(date)}</span>}
                </div>
                {renderCommitList(commits)}
              </div>
            </div>
          );
        })}
      </div>
    </div>
  );
})()}
