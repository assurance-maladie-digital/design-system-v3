export const sectionOrder = {
  'accessibility-demo-section': 20,
  'demo-section': 20,
  'accessibility-keyboard': 30,
  'criteria-section': 40,
  'api-section': 40,
  'implementation-section': 40,
  'adaptive-mode-section': 40,
  'automatic-section': 40,
  'tab-panel-section': 40,
  'best-practices': 50,
  'notice-section': 60,
  'warning-section': 60,
  'audit-section': 70,
  'resources-section': 80,
};

export const getSectionOrder = (node) => {
  if (!isValidElement(node)) return 10;

  const className = typeof node.props.className === 'string' ? node.props.className : '';
  const sectionClass = Object.keys(sectionOrder).find((classNamePart) => className.includes(classNamePart));
  if (sectionClass) return sectionOrder[sectionClass];

  const nestedNode = Children.toArray(node.props.children).find((child) => getSectionOrder(child) !== 10);
  return nestedNode ? getSectionOrder(nestedNode) : 10;
};

export const AccessibilityGuideLayout = ({ componentName, title = `Guide d'Accessibilité du Composant ${componentName}`, a11yVersion, iconSrc, apgHref, apgLabel, className = '', children }) => (
  <div className={`accessibility-guide accessibility-page ${className}`}>
    <div className="header">
      <h1>{title}</h1>
      {(a11yVersion || resolveA11yVersion(componentName)) && (
        <span className="guide-meta">{a11yVersion || resolveA11yVersion(componentName)}</span>
      )}
    </div>

    <div className="intro-section">
      {iconSrc && (
        <img
          src={iconSrc}
          alt="Icône d'accessibilité"
          className="accessibility-icon"
        />
      )}
      <p className="intro-text">
        Le composant {componentName} s'appuie sur les recommandations d'accessibilité du W3C
        {apgHref && (
          <>
            , notamment le modèle{' '}
            <a href={apgHref} target="_blank" rel="noopener noreferrer">
              {apgLabel || 'WAI-ARIA Authoring Practices'}
            </a>
          </>
        )}
        . Ce guide présente les choix d'implémentation, les comportements à vérifier et les bonnes pratiques
        pour une intégration inclusive.
      </p>
    </div>

    {Children.toArray(children)
      .map((child, index) => ({ child, index, order: getSectionOrder(child) }))
      .sort((first, second) => first.order - second.order || first.index - second.index)
      .map(({ child }) => child)}
  </div>
);
